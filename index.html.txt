function show(title, text) {

    const result =
        document.getElementById("result");

    result.style.display = "block";

    result.innerHTML = `

        <h2>${title}</h2>

        <pre>${text}</pre>

    `;

}


function github() {

    window.open(
        "https://github.com/Concerfate",
        "_blank"
    );

}

    show(

`[ DEMONSTRATION ]

Aucune action réelle n'est effectuée.`

}

${navigator.onLine ? "En ligne" : "Hors ligne"}`
    );

}
